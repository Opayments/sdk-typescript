# @opayments/sdk@0.1.0

A TypeScript SDK client for the api.opayments.io API.

## Usage

First, install the SDK from npm.

```bash
npm install @opayments/sdk --save
```

Next, try it out.


```ts
import {
  Configuration,
  BalanceApi,
} from '@opayments/sdk';
import type { GetBalanceRequest } from '@opayments/sdk';

async function example() {
  console.log("🚀 Testing @opayments/sdk SDK...");
  const config = new Configuration({ 
    // To configure API key authorization: RequestSignature
    apiKey: "YOUR API KEY",
    // To configure API key authorization: ProjectIdentity
    apiKey: "YOUR API KEY",
  });
  const api = new BalanceApi(config);

  try {
    const data = await api.getBalance();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Run the test
example().catch(console.error);
```
